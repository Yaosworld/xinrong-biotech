- 热门产品推荐

### 3.3 计算属性

#### 3.3.1 热门产品 (featuredProducts)

```typescript
// 取所有产品排序后的前8个，不受筛选条件影响
const featuredProducts = computed(() => {
  const allProducts = [...productStore.products]

  // 应用排序逻辑
  switch (productStore.sortBy) {
    case 'name-asc':
      allProducts.sort((a, b) => a.name.localeCompare(b.name, 'zh-CN'))
      break
    case 'name-desc':
      allProducts.sort((a, b) => b.name.localeCompare(a.name, 'zh-CN'))
      break
  }

  return allProducts.slice(0, 8)
})
```

**特点：**
- 展示前8个产品
- 支持中文排序（localeCompare）
- 不受产品中心筛选条件影响

#### 3.3.2 推荐品牌 (featuredBrands)

```typescript
// 自有品牌优先，然后是国外知名品牌，总共取前10个
const featuredBrands = computed(() => {
  // 1. 获取自有品牌
  const ownBrands = brandStore.ownBrands
  
  // 2. 获取国外知名品牌（非自有品牌且非中国品牌）
  const internationalBrands = brandStore.brands.filter(brand =>
    brand.is_own_brand !== true && brand.country !== '中国'
  )

  // 3. 合并：自有品牌在前
  const combinedBrands = [...ownBrands, ...internationalBrands]

  // 4. 如果不足10个，补充国内代理品牌
  if (combinedBrands.length < 10) {
    const domesticAgentBrands = brandStore.brands.filter(brand =>
      brand.is_own_brand !== true && brand.country === '中国'
    )
    combinedBrands.push(...domesticAgentBrands)
  }

  // 5. 按排序顺序排序
  combinedBrands.sort((a, b) => (a.sort_order || 999) - (b.sort_order || 999))

  return combinedBrands.slice(0, 10)
})
```

**品牌优先级：**
1. 自有品牌（is_own_brand = true）
2. 国外代理品牌（country !== '中国'）
3. 国内代理品牌（country === '中国'）

#### 3.3.3 最新活动 (latestPromotions)

```typescript
// 按优先级取前8个：即将结束 > 进行中 > 即将开始
const latestPromotions = computed(() => {
  // 使用 store 中已排好序的活动
  const activePromotions = promotionStore.sortedPromotions.filter(p =>
    p.status !== 'ended' && p.status !== 'all'
  )

  return activePromotions.slice(0, 8)
})
```

**活动状态优先级：**
1. `endingSoon` - 即将结束（最紧迫）
2. `active` - 正在进行
3. `coming` - 即将开始
4. 排除 `ended` - 已结束

### 4.3 状态管理

```typescript
const currentImageIndex = ref(0)        // 当前图片索引
const intervalId = ref<number | null>(null)  // 定时器ID
```

### 4.4 核心方法

```typescript
// 下一张
const nextImage = () => {
  currentImageIndex.value = (currentImageIndex.value + 1) % images.length
}

// 上一张
const prevImage = () => {
  currentImageIndex.value = (currentImageIndex.value - 1 + images.length) % images.length
}

// 跳转到指定图片
const goToImage = (index: number) => {
  currentImageIndex.value = index
}

// 开始自动轮播
const startAutoRotation = () => {
  intervalId.value = setInterval(nextImage, 4000)
}

// 停止自动轮播
const stopAutoRotation = () => {
  if (intervalId.value) {
    clearInterval(intervalId.value)
    intervalId.value = null
  }
}
```

### 4.5 生命周期

```typescript
onMounted(() => {
  startAutoRotation()  // 组件挂载时开始轮播
})

onUnmounted(() => {
  stopAutoRotation()   // 组件卸载时清理定时器
})
```

### 4.6 模板结构

```vue
<template>
  <section class="home-banner relative h-[900px] overflow-hidden">
    <!-- 图片轮播容器 -->
    <div class="image-slider" @mouseenter="stopAutoRotation" @mouseleave="startAutoRotation">
      <!-- 图片列表（淡入淡出切换） -->
      <div v-for="(image, index) in images" :key="index"
           :class="{ 'opacity-100': index === currentImageIndex, 'opacity-0': index !== currentImageIndex }">
        <img :src="image" :alt="`Banner ${index + 1}`" />
        <!-- 渐变遮罩 -->
        <div class="bg-gradient-to-b from-black/80 via-black/20 to-black/80"></div>
      </div>

      <!-- 左右切换按钮 -->
      <button @click.stop="prevImage">←</button>
      <button @click.stop="nextImage">→</button>

      <!-- 指示器 -->
      <div class="indicators">
        <button v-for="(_, index) in images" @click.stop="goToImage(index)"
                :class="index === currentImageIndex ? 'w-8 bg-white' : 'bg-white/50'">
        </button>
      </div>
    </div>

    <!-- 内容插槽（搜索框） -->
    <div class="absolute inset-0 flex items-center justify-center z-10">
      <slot></slot>
    </div>
  </section>
</template>
```

### 5.4 搜索高亮

```typescript
// 安全高亮关键词（使用DOMPurify防止XSS）
const highlightText = (text: string) => {
  if (!props.highlightKeyword || !text) return text

  // 分割关键词并转义特殊字符
  const keywords = props.highlightKeyword
    .trim()
    .split(/\s+/)
    .filter(k => k.length > 0)
    .map(k => k.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))

  if (keywords.length === 0) return text

  // 替换匹配的关键词
  let result = text
  keywords.forEach(keyword => {
    const regex = new RegExp(`(${keyword})`, 'gi')
    result = result.replace(regex, '<mark class="search-highlight">$1</mark>')
  })

  // 使用DOMPurify清理HTML
  return DOMPurify.sanitize(result, {
    ALLOWED_TAGS: ['mark'],
    ALLOWED_ATTR: ['class']
  })
}
```

```typescript
const brandName = computed(() => props.brand.name || props.brand.show_name || '')

// 品牌首字母/首字
const brandInitial = computed(() => {
  const name = brandName.value
  // 中文取第一个字
  if (/[\u4e00-\u9fa5]/.test(name.charAt(0))) {
    return name.charAt(0)
  }
  // 英文取首字母大写
  return name.charAt(0).toUpperCase()
})

// 随机背景颜色（根据品牌名生成）
const bgColorClass = computed(() => {
  const colors = [
    'bg-gradient-600', 'bg-purple-500', 'bg-indigo-500',
    'bg-blue-500', 'bg-cyan-500', 'bg-teal-500'
  ]
  const index = brandName.value.charCodeAt(0) % colors.length
  return colors[index]
})
```

### 8.3 搜索逻辑

```typescript
const handleSearch = () => {
  const query = searchInputValue.value.trim()
  if (query) {
    // 带搜索参数跳转到产品中心
    router.push({
      path: '/products',
      query: { search: query }
    })
  } else {
    // 无搜索词直接跳转
    router.push('/products')
  }
}
```
